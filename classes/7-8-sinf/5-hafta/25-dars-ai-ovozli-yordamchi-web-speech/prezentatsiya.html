function processVoiceCommand(spokenText) { // 1. Matnni tozalash va kichik harflarga o'tkazish: const cleanText = spokenText.toLowerCase().trim(); // 2. Niyatlarni (Intents) tekshirish: if (cleanText.includes("soat") || cleanText.includes("vaqt")) { const time = new Date().toLocaleTimeString(); speak(`Hozirgi vaqt: ${time}`); } else if (cleanText.includes("salom")) { speak("Assalomu alaykum, aziz do'stim! Qanday yordam bera olaman?"); } }
OVOZ OHANGI VA TEZLIGI

Jarvis Ovozi: Balandlik (Pitch) va Tezlik (Rate)

Jarvis ovozini xohlagancha sozlash mumkin: robot kabi bas ovoz, o'ta tezkor yoki xotirjam qiroat. SpeechSynthesisUtterance yordamida nutqning tezligi va balandligi boshqariladi.

const msg = new SpeechSynthesisUtterance("Buyruq qabul qilindi, ser!"); msg.pitch = 0.8; // 0.0 dan 2.0 gacha (Pastroq = Yo'g'on robot ovozi) msg.rate = 1.1; // 0.1 dan 10 gacha (Tezlik: 1.0 = Oddiy inson sur'ati) msg.lang = "uz-UZ";// Til kodi (ru-RU, en-US) window.speechSynthesis.speak(msg);

function playJarvisChime() { const ctx = new AudioContext(); const osc = ctx.createOscillator(); const gain = ctx.createGain(); osc.type = "sine"; osc.frequency.setValueAtTime(520, ctx.currentTime); osc.frequency.exponentialRampToValueAtTime(880, ctx.currentTime + 0.15); gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.2); osc.connect(gain); gain.connect(ctx.destination); osc.start(); osc.stop(ctx.currentTime + 0.2); }
