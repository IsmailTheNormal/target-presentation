Har Bir Kristalning Xususiyatlari

  • x, y: Ekranning qaysi piksellarida turishi.
  • size: Hajmi (oddiy 16px, yirik bonus 20px).
  • collected: false: Bayroq: agar qahramon tegsa, true bo'ladi va o'chadi!

Kristalni Yig'ish Kodi

function checkCoinCollisions() {
  coins.forEach(coin => {
    if (!coin.collected &&
        player.x < coin.x + coin.size &&
        player.x + player.w > coin.x &&
        player.y < coin.y + coin.size &&
        player.y + player.h > coin.y) {
      
      coin.collected = true; // Kristall yo'qoladi!
      score += 100;          // 100 ball qo'shiladi!
      playCoinSound();       // Dzing! ovozi yangraydi!
    }
  });
}

Kombi Hisoblash Kodi

let combo = 1;
let comboTimer = null;

function addScoreWithCombo() {
  score += 100 * combo;
  combo++;
  clearTimeout(comboTimer);
  
  // 3 soniyada yangi kristal olinmasa kombi nolga tushadi
  comboTimer = setTimeout(() => { combo = 1; }, 3000);
}

Romb Chizish Kodi

function drawCrystal(x, y, size) {
  ctx.save();
  ctx.fillStyle = "#00e5ff";     // Kiber moviy
  ctx.shadowColor = "#00e5ff";   // Yorug'lik
  ctx.shadowBlur = 12;
  
  ctx.beginPath();
  ctx.moveTo(x, y - size);       // Tepasi
  ctx.lineTo(x + size, y);       // O'ngi
  ctx.lineTo(x, y + size);       // Pasti
  ctx.lineTo(x - size, y);       // Chapi
  ctx.closePath();
  ctx.fill();
  ctx.restore();
}

Neon Porlashi Sirlari
