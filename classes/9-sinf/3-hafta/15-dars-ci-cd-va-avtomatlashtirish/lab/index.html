on:

    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  quality-check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

        with:
          node-version: 20
      - run: npm install
      - run: npm run lint
      - run: npm test

  deploy:
    needs: [ quality-check ]
    runs-on: ubuntu-latest
    steps:
